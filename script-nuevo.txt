/** Busca la fila de una inscripción por su ID. Devuelve {sh, i (fila de la planilla), f} o null. */
function fila_(id) {
  if (!id) return null;
  const sh = hoja_(); const filas = sh.getDataRange().getValues();
  for (let i = 1; i < filas.length; i++) if (String(filas[i][C.id]) === String(id)) return { sh: sh, i: i + 1, f: filas[i] };
  return null;
}

// ---------- rondas.json (una sola fuente para la página y el script) ----------
function rondas_() {
  const cache = CacheService.getScriptCache();
  const c = cache.get('rondas');
  if (c) return JSON.parse(c);
  try {
    const r = UrlFetchApp.fetch(SITIO + '/rondas.json?v=' + Date.now(), { muteHttpExceptions: true });
    if (r.getResponseCode() === 200) {
      const txt = r.getContentText(); const lista = JSON.parse(txt);
      cache.put('rondas', txt, 120);
      PropertiesService.getScriptProperties().setProperty('RONDAS_RESPALDO', txt);
      return lista;
    }
  } catch (e) {}
  return JSON.parse(prop_('RONDAS_RESPALDO') || '[]');
}
function ronda_(id) { return rondas_().filter(r => r.id === id)[0] || null; }

// ---------- Mesa de Tarot ----------
/** true / false si se pudo consultar; null si no hay clave o la consulta falló. */
function tieneMesa_(email) {
  const clave = prop_('RONDA_CLAVE');
  if (!clave || !email) return null;
  try {
    const r = UrlFetchApp.fetch(MESA_FN, { method: 'post', contentType: 'application/json', payload: JSON.stringify({ email: String(email).trim().toLowerCase() }), headers: { 'x-ronda-clave': clave }, muteHttpExceptions: true });
    if (r.getResponseCode() !== 200) return null;
    const j = JSON.parse(r.getContentText());
    return typeof j.tiene === 'boolean' ? j.tiene : null;
  } catch (e) { return null; }
}

// ---------- Mercado Pago ----------
function mp_(path, opt) {
  const t = prop_('MP_TOKEN'); if (!t) return null;
  opt = opt || {};
  const r = UrlFetchApp.fetch('https://api.mercadopago.com' + path, { method: opt.method || 'get', contentType: 'application/json', payload: opt.body ? JSON.stringify(opt.body) : undefined, headers: Object.assign({ Authorization: 'Bearer ' + t }, opt.headers || {}), muteHttpExceptions: true });
  const code = r.getResponseCode();
  if (code < 200 || code >= 300) { console.error('MP ' + path + ' → ' + code + ' ' + r.getContentText().slice(0, 300)); return null; }
  return JSON.parse(r.getContentText());
}
function mpPreferencia_(x) {
  const vuelta = SITIO + '/pagar/?id=' + encodeURIComponent(x.id) + '&vuelta=mp';
  const p = mp_('/checkout/preferences', { method: 'post', headers: { 'X-Idempotency-Key': 'pref-' + x.id + '-' + Date.now() }, body: {
    items: [{ id: x.ronda, title: x.titulo, quantity: 1, unit_price: x.ars, currency_id: 'ARS' }],
    payer: { email: x.email },
    external_reference: x.id,
    back_urls: { success: vuelta, pending: vuelta, failure: vuelta },
    auto_return: 'approved',
    notification_url: EXEC + '?mp=1',
    statement_descriptor: 'RONDA DE TAROT'
  } });
  return p && p.init_point ? p.init_point : null;
}
/** Pago aprobado de Mercado Pago para esta inscripción, o null. */
function mpAprobado_(id, ars) {
  const s = mp_('/v1/payments/search?external_reference=' + encodeURIComponent(id) + '&sort=date_created&criteria=desc');
  if (!s || !s.results) return null;
  const ok = s.results.filter(p => p.status === 'approved' && p.currency_id === 'ARS' && Number(p.transaction_amount) >= ars - 1)[0];
  return ok ? { medio: 'Mercado Pago', monto: ars_(ok.transaction_amount), ref: String(ok.id) } : null;
}

// ---------- PayPal ----------
function ppToken_() {
  const id = prop_('PAYPAL_ID'), sec = prop_('PAYPAL_SECRETO'); if (!id || !sec) return null;
  const cache = CacheService.getScriptCache(); const c = cache.get('pp_token'); if (c) return c;
  const r = UrlFetchApp.fetch('https://api-m.paypal.com/v1/oauth2/token', { method: 'post', payload: { grant_type: 'client_credentials' }, headers: { Authorization: 'Basic ' + Utilities.base64Encode(id + ':' + sec) }, muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) { console.error('PayPal token → ' + r.getResponseCode()); return null; }
  const j = JSON.parse(r.getContentText());
  cache.put('pp_token', j.access_token, Math.max(60, Math.min(21600, (j.expires_in || 3600) - 120)));
  return j.access_token;
}
function pp_(path, opt) {
  const t = ppToken_(); if (!t) return null;
  opt = opt || {};
  const r = UrlFetchApp.fetch('https://api-m.paypal.com' + path, { method: opt.method || 'get', contentType: 'application/json', payload: opt.body !== undefined ? JSON.stringify(opt.body) : undefined, headers: Object.assign({ Authorization: 'Bearer ' + t }, opt.headers || {}), muteHttpExceptions: true });
  const code = r.getResponseCode();
  if (code < 200 || code >= 300) { console.error('PayPal ' + path + ' → ' + code + ' ' + r.getContentText().slice(0, 300)); return null; }
  return JSON.parse(r.getContentText() || '{}');
}
/** Crea una orden de PayPal y devuelve {orden, url}. */
function ppOrden_(x) {
  const vuelta = SITIO + '/pagar/?id=' + encodeURIComponent(x.id) + '&vuelta=pp';
  const o = pp_('/v2/checkout/orders', { method: 'post', headers: { 'PayPal-Request-Id': 'ord-' + x.id + '-' + Date.now() }, body: {
    intent: 'CAPTURE',
    purchase_units: [{ reference_id: x.id, custom_id: x.id, description: x.titulo, amount: { currency_code: 'USD', value: x.usd.toFixed(2) } }],
    payment_source: { paypal: { experience_context: { brand_name: 'Ronda de Tarot', shipping_preference: 'NO_SHIPPING', user_action: 'PAY_NOW', return_url: vuelta, cancel_url: vuelta + '&cancelado=1' } } }
  } });
  if (!o || !o.id) return null;
  const l = (o.links || []).filter(k => k.rel === 'payer-action' || k.rel === 'approve')[0];
  return l ? { orden: o.id, url: l.href } : null;
}
/** Revisa (y si hace falta cobra) una orden de PayPal. Devuelve el pago o null. */
function ppAprobado_(orden, id, usd) {
  let o = pp_('/v2/checkout/orders/' + encodeURIComponent(orden));
  if (!o) return null;
  const pu = (o.purchase_units || [])[0] || {};
  if (pu.custom_id && pu.custom_id !== id) return null;
  if (o.status === 'APPROVED') o = pp_('/v2/checkout/orders/' + encodeURIComponent(orden) + '/capture', { method: 'post', body: {}, headers: { 'PayPal-Request-Id': 'cap-' + orden } }) || o;
  const cap = ((((o.purchase_units || [])[0] || {}).payments || {}).captures || [])[0];
  if (!cap || cap.status !== 'COMPLETED') return null;
  if (cap.amount.currency_code !== 'USD' || Number(cap.amount.value) < usd - 0.01) return null;
  return { medio: 'PayPal', monto: 'US$ ' + Number(cap.amount.value).toString().replace('.', ','), ref: cap.id };
}

function titulo_(f) { return 'Ronda de Tarot ' + (f[C.tipo] === 'online' ? 'online' : 'presencial') + ' · ' + fechaLarga_(String(f[C.fRonda])) + (f[C.cobro] === 'sena' ? ' (seña)' : ''); }

/** Arma el cobro de una inscripción (desde la página o desde el mail) y devuelve a dónde ir. */
function pagar_(id, medio) {
  const x = fila_(id);
  if (!x) return { ok: false, msg: 'No encontramos tu inscripción.' };
  const f = x.f;
  if (f[C.estado] === 'confirmada') return { ok: true, pagado: true };
  if (f[C.estado] === 'cancelada') return { ok: false, msg: 'Esta inscripción está cancelada. Si es un error, escribinos.' };
  const pago = revisarPago_(x); if (pago) return { ok: true, pagado: true };
  const r = ronda_(f[C.ronda]) || {};
  const base = { id: f[C.id], ronda: f[C.ronda], titulo: titulo_(f), email: f[C.mail], ars: Number(f[C.ars]) || r.ars || num_(r.monto), usd: Number(f[C.usd]) || r.usd || 0 };
  if (medio === 'pp') {
    if (!base.usd) return { ok: false, msg: 'Esta ronda no se paga con PayPal.' };
    const o = ppOrden_(base);
    if (o) {
      x.sh.getRange(x.i, C.ordenes + 1).setValue([f[C.ordenes], o.orden].filter(String).join(','));
      return { ok: true, url: o.url };
    }
    return r.pago_usd ? { ok: true, url: r.pago_usd, fijo: true } : { ok: false, msg: 'No pudimos armar el pago. Probá de nuevo en un rato.' };
  }
  const u = mpPreferencia_(base);
  if (u) return { ok: true, url: u };
  return r.pago ? { ok: true, url: r.pago, fijo: true } : { ok: false, msg: 'No pudimos armar el pago. Probá de nuevo en un rato.' };
}

/** Lo consulta la página a la vuelta del pago. */
function estado_(id, orden) {
  let x = fila_(id);
  if (!x) return { ok: false, msg: 'No encontramos tu inscripción.' };
  if (orden && PENDIENTES.indexOf(x.f[C.estado]) >= 0 && String(x.f[C.ordenes]).split(',').indexOf(String(orden)) < 0) {
    // la orden vuelve en la URL de PayPal; solo se acepta si es de esta inscripción (custom_id)
    const o = pp_('/v2/checkout/orders/' + encodeURIComponent(orden));
    if (o && ((o.purchase_units || [])[0] || {}).custom_id === x.f[C.id]) x.sh.getRange(x.i, C.ordenes + 1).setValue([x.f[C.ordenes], orden].filter(String).join(','));
    x = fila_(id);
  }
  revisarPago_(x);
  x = fila_(id);
  return { ok: true, estado: x.f[C.estado], pagado: x.f[C.estado] === 'confirmada', nombre: String(x.f[C.nombre]).trim().split(' ')[0], tipo: x.f[C.tipo], cobro: x.f[C.cobro] };
}

/** Si la inscripción está pendiente y el pago ya se acreditó, la confirma. Devuelve el pago o null. */
function revisarPago_(x) {
  if (!x || PENDIENTES.indexOf(x.f[C.estado]) < 0) return null;
  const f = x.f; const id = String(f[C.id]); if (!id) return null;
  let pago = null;
  const ars = Number(f[C.ars]) || 0, usd = Number(f[C.usd]) || 0;
  if (ars) pago = mpAprobado_(id, ars);
  if (!pago && usd) String(f[C.ordenes] || '').split(',').filter(String).some(o => (pago = ppAprobado_(o, id, usd)));
  if (pago) confirmar_(x, pago);
  return pago;
}

function confirmar_(x, pago) {
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const actual = x.sh.getRange(x.i, C.estado + 1).getValue();
    if (PENDIENTES.indexOf(actual) < 0) return;
    const sena = x.f[C.cobro] === 'sena';
    x.sh.getRange(x.i, C.estado + 1).setValue('confirmada');
    x.sh.getRange(x.i, C.pago + 1).setValue(pago.gratis ? 'sin cargo (tiene Mesa)' : pago.medio + ' · ' + pago.monto + (sena ? ' (seña)' : '') + ' · #' + pago.ref + ' · ' + Utilities.formatDate(new Date(), 'America/Argentina/Buenos_Aires', 'dd/MM HH:mm'));
    if (pago.gratis) x.sh.getRange(x.i, C.mesa + 1).setValue('sí (verificado)');
  } finally { lock.releaseLock(); }
  const f = x.f; const r = ronda_(f[C.ronda]) || {};
  const m = datosMail_(Object.assign({ id: f[C.ronda], tipo: f[C.tipo], fecha: String(f[C.fRonda]), hora: String(f[C.hora]), fin: String(f[C.fin]), lugar: f[C.lugar], precio: f[C.precio] }, r),
    { nombre: f[C.nombre], email: f[C.mail], grabacion: f[C.grab], tz: f[C.tz], pais: f[C.pais], id: f[C.id], cobra: false, gratis: !!pago.gratis });
  m.cobro = f[C.cobro]; m.pagado = pago;
  enviar_('pago', m);
  MailApp.sendEmail(AVISOS_A, (pago.gratis ? 'Mesa verificada · ' : 'Pago recibido · ') + f[C.ronda] + ' · ' + f[C.nombre],
    f[C.nombre] + ' · ' + f[C.mail] + '\n' + (pago.gratis ? 'Tiene Mesa de Tarot: queda confirmada sin cargo.' : pago.medio + ' · ' + pago.monto + ' · #' + pago.ref) + '\n\nPanel: ' + SITIO + '/admin/');
}

/** Aviso de Mercado Pago (notification_url = EXEC?mp=1). */
function avisoMp_(e) {
  try {
    const p = e.parameter || {};
    let pid = p['data.id'] || (p.topic === 'payment' || p.type === 'payment' ? p.id : '');
    if (!pid && e.postData && e.postData.contents) { const b = JSON.parse(e.postData.contents); if (b && (b.type === 'payment' || /^payment/.test(b.action || ''))) pid = b.data && b.data.id; }
    if (!pid) return { ok: true };
    const pg = mp_('/v1/payments/' + encodeURIComponent(pid));
    if (pg && pg.external_reference) revisarPago_(fila_(pg.external_reference));
  } catch (err) { console.error('avisoMp_', err); }
  return { ok: true };
}

/** Corre solo cada 15 minutos (instalar una vez con instalarAvisos). */
function enviarAvisos() {
  revisarPendientes_();
  const sh = hoja_(); const filas = sh.getDataRange().getValues(); const ahora = new Date();
  for (let i = 1; i < filas.length; i++) {
    const f = filas[i]; if (f[C.estado] !== 'confirmada') continue;
    const d = { ronda: f[C.ronda], tipo: f[C.tipo], nombre: f[C.nombre], email: f[C.mail], grabacion: f[C.grab], fecha: String(f[C.fRonda]), hora: String(f[C.hora]), fin: String(f[C.fin]), lugar: f[C.lugar], precio: f[C.precio], tz: f[C.tz] || '' };
    const inicio = new Date(d.fecha + 'T' + d.hora + ':00-03:00');
    const falta = (inicio - ahora) / 60000; // minutos
    if (!f[C.avisoDia] && falta <= 24 * 60 && falta > MIN_ANTES) { enviar_('dia_antes', d); sh.getRange(i + 1, C.avisoDia + 1).setValue(new Date()); }
    if (d.tipo === 'online' && !f[C.avisoPrevio] && falta <= MIN_ANTES && falta > -15) { enviar_('previo', d); sh.getRange(i + 1, C.avisoPrevio + 1).setValue(new Date()); }
  }
}
/** Pagos pendientes y Mesa sin verificar de rondas que todavía no pasaron. */
function revisarPendientes_() {
  const sh = hoja_(); const filas = sh.getDataRange().getValues(); const ahora = new Date();
  for (let i = 1; i < filas.length; i++) {
    const f = filas[i];
    if (PENDIENTES.indexOf(f[C.estado]) < 0 || !f[C.id]) continue;
    if (new Date(String(f[C.fRonda]) + 'T23:59:00-03:00') < ahora) continue;
    const x = { sh: sh, i: i + 1, f: f };
    try {
      if (f[C.mesa] === 'dijo que sí, sin verificar' && tieneMesa_(f[C.mail]) === true) { confirmar_(x, { gratis: true }); continue; }
      revisarPago_(x);
    } catch (err) { console.error('revisarPendientes_', err); }
  }
}
function instalarAvisos() {
  ScriptApp.getProjectTriggers().forEach(t => { if (t.getHandlerFunction() === 'enviarAvisos') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('enviarAvisos').timeBased().everyMinutes(15).create();
}

}

function adminEstado(id, estado) {
  soloLo_();
  if (['confirmada', 'seña pendiente', 'pago pendiente', 'cancelada'].indexOf(estado) < 0) throw new Error('Estado inválido');
  const x = fila_(id); if (!x) throw new Error('No encontré la inscripción');
  x.sh.getRange(x.i, C.estado + 1).setValue(estado);
  if (estado === 'confirmada' && /^pendiente/.test(String(x.f[C.pago]))) x.sh.getRange(x.i, C.pago + 1).setValue(String(x.f[C.pago]) + ' · confirmada a mano');
  const f = fila_(id).f; return { estado: f[C.estado], pago: String(f[C.pago]) };
}
function adminRevisar(id) {
  soloLo_();
  const x = fila_(id); if (!x) throw new Error('No encontré la inscripción');
  const pago = revisarPago_(x);
  const f = fila_(id).f; return { encontrado: !!pago, estado: f[C.estado], pago: String(f[C.pago]) };
}
function adminMesa(id) {
  soloLo_();
  const x = fila_(id); if (!x) throw new Error('No encontré la inscripción');
  const tiene = tieneMesa_(x.f[C.mail]);
  if (tiene === true && PENDIENTES.indexOf(x.f[C.estado]) >= 0) confirmar_(x, { gratis: true });
  else if (tiene === true) x.sh.getRange(x.i, C.mesa + 1).setValue('sí (verificado)');
  else if (tiene === false && x.f[C.mesa]) x.sh.getRange(x.i, C.mesa + 1).setValue('dijo que sí, no encontrado');
  const f = fila_(id).f; return { tiene: tiene, estado: f[C.estado], mesa: String(f[C.mesa]), pago: String(f[C.pago]) };
}
